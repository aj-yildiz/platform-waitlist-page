/**
 * Google Apps Script to save Vastis waitlist and space suggestions to Google Sheets
 * 
 * Sheet Structure:
 * 1. "Vastis_Waitlist" sheet columns:
 *    - Email (A)
 *    - User_Type (B) - Values: Patient, Practitioner, Gym
 *    - Timestamp (C) - ISO format
 *    - Source (D) - Default: 'Website'
 * 
 * 2. "Vastis_Spaces" sheet columns:
 *    - Email (A)
 *    - Space_Suggestion (B)
 *    - Timestamp (C) - ISO format
 *    - Source (D) - Default: 'Website'
 * 
 * Setup Instructions:
 * 1. Create a new Google Sheet named "Vastis Platform Waitlist"
 * 2. Create two sheets: "Vastis_Waitlist" and "Vastis_Spaces"
 * 3. Go to Extensions > Apps Script
 * 4. Replace the default code with this script
 * 5. Deploy as a web app:
 *    - Click Deploy > New deployment
 *    - Select type: Web app
 *    - Description: "Vastis Waitlist Handler"
 *    - Execute as: "Me"
 *    - Who has access: "Anyone"
 *    - Click Deploy
 *    - Copy the web app URL
 * 6. Add the URL as GOOGLE_SCRIPT_URL in your .env file
 */

function validateEmail(email) {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return emailRegex.test(email);
}

function initializeSheet(ss, sheetName, headers) {
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
    // Freeze the header row
    sheet.setFrozenRows(1);
    // Auto-resize columns
    sheet.autoResizeColumns(1, headers.length);
  }
  return sheet;
}

function doPost(e) {
  try {
    // Parse the request body
    const data = e.postData && e.postData.contents ? 
      JSON.parse(e.postData.contents) : e.parameter;
    
    // Validate required fields
    if (!data.email) {
      throw new Error('Email is required');
    }
    if (!validateEmail(data.email)) {
      throw new Error('Invalid email format');
    }

    const timestamp = data.timestamp || new Date().toISOString();
    const source = data.source || 'Website';
    const formType = data.formType || 'waitlist';
    const email = data.email.toLowerCase().trim();
    
    // Get the active spreadsheet
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Handle different form types
    if (formType === 'space_suggestion') {
      if (!data.spaceSuggestion) {
        throw new Error('Space suggestion is required');
      }
      
      const sheet = initializeSheet(ss, 'Vastis_Spaces', 
        ['Email', 'Space_Suggestion', 'Timestamp', 'Source']);
      
      sheet.appendRow([
        email,
        data.spaceSuggestion,
        timestamp,
        source
      ]);
      
    } else {
      // Waitlist submission
      if (!data.userType) {
        throw new Error('User type is required');
      }
      if (!['Patient', 'Practitioner', 'Gym'].includes(data.userType)) {
        throw new Error('Invalid user type. Must be Patient, Practitioner, or Gym');
      }
      
      const sheet = initializeSheet(ss, 'Vastis_Waitlist',
        ['Email', 'User_Type', 'Timestamp', 'Source']);
      
      sheet.appendRow([
        email,
        data.userType,
        timestamp,
        source
      ]);
    }
    
    // Log successful submission
    Logger.log("Successfully saved submission: " + JSON.stringify({
      email,
      formType,
      timestamp,
      source
    }));
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: 'Data saved successfully',
      formType: formType
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    Logger.log("Error processing submission: " + error.toString());
    
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet() {
  return ContentService.createTextOutput(JSON.stringify({
    status: 'active',
    message: 'Vastis Waitlist API is active. Use POST to submit data.',
    version: '1.0.0'
  })).setMimeType(ContentService.MimeType.JSON);
}
